# Совместимость тайлсетов (tileset merge compatibility)
#
# Формат строки:  ИмяA, ИмяB, true|false
#   Правило задаётся ДЛЯ ПАРЫ и работает симметрично: если Земля сливается с Камнем,
#   то Камень сливается с Землёй. Каждую пару достаточно указать один раз.
#   Missing rule = слияния нет (клетки рисуются с границей) — чтобы отключить пару,
#   достаточно поставить false или удалить строку.
#
# Понятные имена (алиасы) загружаются из SpriteData: можно писать имена спрайтов
# (Tiles_0, Tiles_1, Tiles_8, Tiles_coal, …) или короткие слова ниже.
# Допустимы и точные имена типов: Terrain0, TerrainGrass, Stone, Terrain8, …
#
#   Земля = Terrain0 (Tiles_0)      Трава = TerrainGrass (Tiles_2)
#   Камень = Stone (Tiles_1)        Руда = Terrain8 (Tiles_8)
#
# Файл читается один раз при первом использовании.
# Изменения после загрузки применяются на следующем запуске игры.

Земля,Камень,true
Земля,Руда,true
Камень,Руда,false
Камень,Трава,true
Руда,Трава,true
